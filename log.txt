      }
    } else {
      Logger.log("BOOT", "Control panel auto-mount skipped by Config.bootGui.autoMountPanel");
    }
    
    if (!window.__ligmarBotDiagHandlersInstalled) {
      window.__ligmarBotDiagHandlersInstalled = true;
      window.addEventListener("error", (ev) => {
        try { createDiagnosticReport("window_error", { message: ev.message, filename: ev.filename, lineno: ev.lineno, colno: ev.colno, error: ev.error && (ev.error.stack || String(ev.error)) }); } catch (err) {}
      });
      window.addEventListener("unhandledrejection", (ev) => {
        try {
          const rawReason = ev && ev.reason ? (ev.reason.stack || String(ev.reason)) : "";
          const now = Date.now();
          const isClipboardNoise = String(rawReason || "").indexOf("Clipboard") !== -1 || String(rawReason || "").indexOf("writeText") !== -1;
          if (isClipboardNoise && ev && typeof ev.preventDefault === "function") {
            ev.preventDefault();
          }
          if (!window.__ligmarBotLastUnhandledReportAt || now - window.__ligmarBotLastUnhandledReportAt > 5000) {
            window.__ligmarBotLastUnhandledReportAt = now;
            createDiagnosticReport("unhandled_rejection", { reason: rawReason }, { autoCopy: false });
          } else if (Logger && Logger.debug) {
            Logger.debug("DIAG", "Suppressed repeated unhandled rejection report", { reason: rawReason });
          }
        } catch (err) {}
      });
    }

      const snapshotTickMs = (Config.timings && Number.isFinite(Config.timings.stateSnapshotTickMs))
        ? Config.timings.stateSnapshotTickMs
        : 1000;
      setInterval(() => {
        const state = readBasicState();
        Logger.log("STATE", "Basic snapshot", state);
      }, snapshotTickMs);
    }
